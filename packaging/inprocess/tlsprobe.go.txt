package main

import (
	"crypto/tls"
	"crypto/x509"
	"fmt"
	"os"
)

func main() {
	pool, err := x509.SystemCertPool()
	fmt.Println("pool", pool != nil, err)
	d := &tls.Dialer{Config: &tls.Config{ServerName: os.Args[1], RootCAs: pool}}
	c, err := d.Dial("tcp", os.Args[1]+":443")
	fmt.Println("dial", err)
	if c != nil { c.Close() }
	d2 := &tls.Dialer{Config: &tls.Config{ServerName: os.Args[1]}}
	c, err = d2.Dial("tcp", os.Args[1]+":443")
	fmt.Println("dial nil roots", err)
}
